  \begin{frame}[t]{Correctness criterion: what must the CFG preserve?}
    \vspace{0.2cm}
    \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\textwidth]{apisurface}
      {\small\bfseries\color{apiblue}Surface fragment $G'$}
      \hfill
      {\footnotesize e.g., names, invocation, member chains}

      \vspace{0.4em}
      \centering
      \fontsize{10.0}{13.6}\selectfont
      \hspace{0.1cm}$\displaystyle
        S ::= E \texttt{;}
        \:\:\:
        E ::= \texttt{ID}\mid\texttt{ID()}\mid\texttt{ID(\,ARGS\,)}\mid E\texttt{.}E
        \:\:\:
        \mathtt{ARGS} ::= E\mid E\texttt{,}\,\texttt{ARGS}
      $
      \vspace{0.15em}
    \end{beamercolorbox}

    \vspace{-0.3em}
    \begin{columns}[T,onlytextwidth]
      \column{0.48\textwidth}
        \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\linewidth]{apiprojection}
          {\small\bfseries\color{apiblue}1. Surface projection $\pi_\Sigma$}

          \vspace{0.1em}
          \centering
          {\scriptsize erase names but preserve punctuation}

          \vspace{0.3em}
          \resizebox{0.88\linewidth}{!}{$
            \pi_\Sigma(\code{x.f(y)})=\texttt{ID.ID(ID)}
          $}

          \vspace{0.3em}
          {\scriptsize checks that the shape belongs to $G_0$}
        \end{beamercolorbox}
      \column{0.48\textwidth}
        \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\linewidth]{apiprojection}
          {\small\bfseries\color{apiteal}2. Typed projection $\Pi_\Gamma$}

          \vspace{0.1em}
          \centering
          {\scriptsize names $\to$ type slots; API names stay}

          \vspace{0.3em}
          \resizebox{0.88\linewidth}{!}{$
            \Pi_\Gamma(\code{x.f(y)})\ni
              \exprnt{\tstate{X}}\texttt{.f(}\exprnt{\tstate{Y}}\texttt{)}
          $}

          \vspace{0.3em}
          {\scriptsize contains all and only well-typed terms}
        \end{beamercolorbox}
    \end{columns}

    \vspace{0.4em}
    \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\textwidth]{apiguarantee}
      {\small\bfseries\color{apiteal}Correctness criterion}

      \vspace{0.15em}

      {\footnotesize For a target language $L$ with surface fragment $G'$, construct $G_\Gamma$ satisfying:}

      \vspace{0.5em}
      \hspace{0.6cm}\resizebox{0.91\linewidth}{!}{$
        \pi_\Sigma(\sigma) \in \mathcal{L}(G')
        \quad\Longleftrightarrow\quad
        \Gamma \vdash \Pi_\Gamma(\sigma) \in \mathcal{L}(G_\Gamma)
      $}

      \vspace{0.4em}
      {\footnotesize for all source code statements $\sigma$ accepted by the reference compiler for $L$.}

    \end{beamercolorbox}

  \begin{frame}[t]{Monomorphization I: ground rules}
    \vspace{-0.45em}
    \begin{beamercolorbox}[rounded=true,sep=0.3em,wd=\textwidth]{apisurface}
      \centering\scriptsize
      \textcolor{apiblue}{\bfseries State $\to$ nonterminal;}\quad
      API text $\to$ terminal;\quad validated AST node $\to$ production.
    \end{beamercolorbox}

    \vspace{0.4em}
    \begin{columns}[T,onlytextwidth]
      \column{0.485\textwidth}
        \begin{beamercolorbox}[rounded=true,sep=0.35em,wd=\linewidth]{apiruleblue}
          {\small\bfseries\color{apiblue}Monomorphic member call}

          \vspace{0.05em}
          \centering
          \resizebox{0.82\linewidth}{!}{$\displaystyle
            \frac{
              \substack{
                \Delta(\tstate{C}\!::\!f)=(\tstate{A})\to\tstate{R}\\[-0.1em]
                \Delta;\Gamma\vdash e_0:\tstate{C}\quad
                \Delta;\Gamma\vdash e_1:\tstate{A}
              }
            }{
              \Delta;\Gamma\vdash e_0\mathtt{.f(}e_1\mathtt{)}:\tstate{R}
            }
          $}

          \vspace{0.1em}
          {\scriptsize becomes}
          \[
            \boxed{\exprnt{\tstate{R}}\to
              \exprnt{\tstate{C}}\code{. f (}\exprnt{\tstate{A}}\code{)}}
          \]
          \vspace{-0.55em}
          {\scriptsize\bfseries Example}\quad
          {\fontsize{6.8}{8.2}\selectfont\ttfamily C c; A a; c.f(a);}
        \end{beamercolorbox}
      \column{0.485\textwidth}
        \begin{beamercolorbox}[rounded=true,sep=0.35em,wd=\linewidth]{apirulegreen}
          {\small\bfseries\color{apiteal}Resolved operator call}

          \vspace{0.05em}
          \centering
          \resizebox{0.82\linewidth}{!}{$\displaystyle
            \frac{
              \substack{
                \Delta;\Gamma\vdash e_1:\tstate{T}_1\quad
                \Delta;\Gamma\vdash e_2:\tstate{T}_2\\[-0.1em]
                \delta^{\Delta}_{+}(\tstate{T}_1,\tstate{T}_2)=\tstate{R}
              }
            }{
              \Delta;\Gamma\vdash e_1\mathtt{+}e_2:\tstate{R}
            }
          $}

          \vspace{0.1em}
          {\scriptsize becomes}
          \[
            \boxed{\exprnt{\tstate{R}}\to
              \exprnt{\tstate{T}_1}\code{+}\exprnt{\tstate{T}_2}}
          \]
          \vspace{-0.55em}
          {\scriptsize\bfseries Example}\quad
          {\fontsize{6.8}{8.2}\selectfont\ttfamily Vec u, v; u + v;}
        \end{beamercolorbox}
    \end{columns}

    \vspace{0.05em}
    \begin{beamercolorbox}[rounded=true,sep=0.2em,wd=\textwidth]{apiwarning}
      \centering
      \resizebox{0.92\linewidth}{!}{$
        \mathrm{AddClosed}(T)\;\Longleftrightarrow\;
          \mathrm{base}\!\left(\delta^{\Delta}_{+}(\tstate{T},\tstate{T})\right)=T
        \quad\overset{\text{erase state}}{\Longrightarrow}\quad
        \boxed{\exprnt{T}\to\exprnt{T}\code{+}\exprnt{T}}
      $}

      \vspace{0.02em}
      {\tiny Type-level mnemonic only; $G_0^+$ preserves C++ precedence/left associativity,
        and mixed states use $\delta_+^\Delta$.}
    \end{beamercolorbox}
  \end{frame}

  \begin{frame}[t]{Monomorphization II: subtype calls}
    \vspace{-0.45em}
    \begin{beamercolorbox}[rounded=true,sep=0.35em,wd=\textwidth]{apisurface}
      \centering\scriptsize
      $D<:_\Delta B$ is public, unambiguous inheritance;
      whole-call resolution retains cv/ref/category state and conversions.
    \end{beamercolorbox}

    \vspace{0.4em}
    \begin{columns}[T,onlytextwidth]
      \column{0.47\textwidth}
        \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\linewidth]{apiruleblue}
          {\small\bfseries\color{apiblue}Resolve the subtype call}

          \centering
          \vspace{0.35em}
          \resizebox{0.96\linewidth}{!}{$\displaystyle
            \frac{
              \substack{
                D<:_\Delta B
                \quad \Delta;\Gamma\vdash e:\tstate{D}\\[-0.1em]
                \mathrm{resolve}_{\Delta}(f;\tstate{D})=((\tstate{B})\to\tstate{R})
              }
            }{
              \Delta;\Gamma\vdash f(e):\tstate{R}
            }\;\rulename{Inv$_{<:}$}
          $}

          \vspace{0.3em}
          {\scriptsize\color{apiblue}
            No global $e:B$ rule: invocation consumes the subtype fact.}

          \vspace{0.35em}
          {\fontsize{7.4}{8.8}\selectfont
          \begin{tabular}{@{}l@{\hspace{0.7em}}>{\ttfamily}l@{}}
            \bfseries Example & struct Animal \{\}; \\
              & struct Dog : Animal \{\}; \\
              & int age(const Animal\&); \\
              & Dog d; age(d); \\
          \end{tabular}}
          \vspace{0.8em}
        \end{beamercolorbox}
      \column{0.49\textwidth}
        \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\linewidth]{apirulegreen}
          {\small\bfseries\color{apiteal}Emit the exact surface call}

          \vspace{0.35em}
          \centering
          \resizebox{0.94\linewidth}{!}{$\displaystyle
            \frac{
              \mathrm{resolve}_{\Delta}(f;\tstate{D})=((\tstate{B})\to\tstate{R})
            }{
              \bigl(\exprnt{\tstate{R}}\to
                \code{f (}\exprnt{\tstate{D}}\code{)}\bigr)\in P_{\mathrm{api},b}
            }\;\rulename{Inv$_\varphi$}
          $}

          \vspace{0.35em}
          {\scriptsize State hats are elided below.}
          \[
            \boxed{\exprnt{\mathtt{int}}\to
              \code{age (}\exprnt{\mathtt{Dog}}\code{)}}
          \]
          \vspace{-0.2em}
          \resizebox{0.84\linewidth}{!}{$
            \begin{aligned}
              \exprnt{\mathtt{int}}&\Rightarrow
                \code{age (}\exprnt{\mathtt{Dog}}\code{)},\\[-0.05em]
              \code{age (}\exprnt{\mathtt{Dog}}\code{)}
                &\in\Pi_\Gamma(\texttt{age(d)}).
            \end{aligned}
          $}
        \end{beamercolorbox}
    \end{columns}

    \vspace{0.05em}
    \begin{beamercolorbox}[rounded=true,sep=0.2em,wd=\textwidth]{apiwarning}
      \centering
      {\scriptsize
        \textbf{Whole-call resolution} preserves overloads, access,
        cv/ref/category, and ambiguity.}

      \vspace{0.05em}
      {\scriptsize
        $\mathtt{Dog}<:\mathtt{Animal}$ does not lift through
        $\mathtt{vector}\langle\cdot\rangle$ (class-template invariance).}
    \end{beamercolorbox}
  \end{frame}

  \begin{frame}[t]{Monomorphizing templates}
    \vspace{-0.55em}
