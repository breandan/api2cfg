        S ::= E \texttt{;}
        \:\:\:

      $
      \vspace{0.15em}

          \vspace{0.3em}

      \vspace{0.15em}

      {\footnotesize For a target language $L$ with surface fragment $G'$, construct $G_\Gamma$ satisfying}

      \vspace{0.5em}
      \hspace{0.6cm}\resizebox{0.91\linewidth}{!}{$

          \vspace{0.35em}
          \centering
          {\fontsize{8.1}{10.2}\selectfont
          $\mathtt{f(A)}:\mathtt{D}\to\mathtt{A}$
          \qquad
          $\mathtt{f(B)}:\mathtt{D}\to\mathtt{B}$}

          \vspace{0.5em}
          {\footnotesize Both candidates are viable by conversion but unresolvable.}

          \vspace{0.7em}
          {\fontsize{9}{9.8}\selectfont
          $\displaystyle
            \operatorname{resolve}_{\Gamma}\big(\code{f(d)}\big) = \:\:?$}

          \vspace{0.7em}
        \end{beamercolorbox}
    \end{columns}

    \nointerlineskip
    \begin{beamercolorbox}[rounded=true,sep=0.4em,wd=\textwidth]{apiwarning}
      {\small\bfseries\color{apiamber}Caveat: local subtype closure is unsound}

      \vspace{0.5em}
      \centering
      {\fontsize{9}{9.4}\selectfont
      $\displaystyle
        \left.
        \begin{aligned}
          \exprnt{\mathtt{int}}&\to\code{f(}\exprnt{X\,}\code{)}\\
          \exprnt{X\,}&\to\exprnt{\mathtt D}
        \end{aligned}
        \right\}
        X\in\{\mathtt A,\mathtt B\}
        \Rightarrow_{G_\Gamma}
        \exprnt{\mathtt{int}}\Rightarrow^*
          \code{f(}\exprnt{\mathtt D}\code{)}$
      }

      \vspace{0.5em}
      \raggedright
      {\footnotesize Local subtype closure derives
      \texttt{f(}$\exprnt{\mathtt D}$\texttt{)}, but \CC rejects $\code{f(d)}$ as ambiguous.}

      \vspace{0.2em}
      {\footnotesize\textcolor{apiteal}{\bfseries Resolve before lowering.}
      Emit a ground call production only when the exact call has a unique best
      viable function; ambiguous calls yield no productions.}
    \end{beamercolorbox}
  \end{frame}
